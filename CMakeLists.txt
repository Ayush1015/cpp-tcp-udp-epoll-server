cmake_minimum_required(VERSION 3.16)
project(epoll_network_server VERSION 1.0 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Linux is required (epoll/eventfd)")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
option(ENABLE_TSAN "Build with ThreadSanitizer" OFF)
add_library(network src/server.cpp src/thread_pool.cpp)
target_include_directories(network PUBLIC include)
target_link_libraries(network PUBLIC Threads::Threads)
target_compile_options(network PUBLIC -Wall -Wextra -Wpedantic -Werror)
if(ENABLE_TSAN)
  target_compile_options(network PUBLIC -fsanitize=thread -g -O1 -fno-pie)
  target_link_options(network PUBLIC -fsanitize=thread -no-pie)
endif()
add_executable(network_server src/main.cpp)
target_link_libraries(network_server PRIVATE network)
include(CTest)
if(BUILD_TESTING)
  if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/catch.hpp")
    file(DOWNLOAD "https://raw.githubusercontent.com/catchorg/Catch2/v2.13.10/single_include/catch2/catch.hpp" "${CMAKE_CURRENT_BINARY_DIR}/catch.hpp" EXPECTED_HASH SHA256=3725c0f0a75f376a5005dde31ead0feb8f7da7507644c201b814443de8355170 TLS_VERIFY ON)
    set(CATCH_DIR "${CMAKE_CURRENT_BINARY_DIR}")
  else()
    set(CATCH_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party")
  endif()
  add_executable(network_tests tests/test_thread_pool.cpp tests/test_network.cpp)
  target_include_directories(network_tests SYSTEM PRIVATE "${CATCH_DIR}")
  target_link_libraries(network_tests PRIVATE network)
  add_test(NAME network_tests COMMAND network_tests)
  set_tests_properties(network_tests PROPERTIES TIMEOUT 60)
endif()
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME cli_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py $<TARGET_FILE:network_server>)
  set_tests_properties(cli_integration PROPERTIES TIMEOUT 30)
endif()
